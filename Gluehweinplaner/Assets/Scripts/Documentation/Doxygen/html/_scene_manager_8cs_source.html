359 public List<Plate> ReserveBudenPosition(Bude b, Vector3 _start, Vector3 _end)
360 {
361 Vector2Int start = WorldPositionToPlateArrayPosition(_start);
362 Vector2Int end = WorldPositionToPlateArrayPosition(_end);
363 Vector2Int dirEnd = end - start;
364 Vector3 worldEnd = _end - _start;
365 List<Vector2Int> platePosToVisit = GenerateMatrix.InterpolateArray(start, end, ((Vector2Int a, Vector2Int b) toCompare) => (Vector3.Distance(allPlateArray![toCompare.a.x, toCompare.a.y].Center, _end) < Vector3.Distance(allPlateArray[toCompare.b.x, toCompare.b.y].Center, _end)) ? toCompare.a : toCompare.b, false, plateCountX, plateCountZ);
366 List<Plate> platesToVisit = new List<Plate>();
367 foreach (Vector2Int pos in platePosToVisit)
368 {
369 platesToVisit.Add(allPlateArray![pos.x, pos.y]);
370 }
371 Vector2Int startPos = allPlateArray![platePosToVisit[0].x, platePosToVisit[0].y].GetPositionInArray(_start, true);
372 if (platePosToVisit.Count > 1)
373 {
374 Vector2Int dirToNextPlate = platePosToVisit[1] - platePosToVisit[0];
375 for (int i = 0; i < platePosToVisit.Count; i++)
376 {
377 if (i < platePosToVisit.Count - 1)
378 {
379 dirEnd = end - platePosToVisit[i];
380 startPos = allPlateArray[platePosToVisit[i].x, platePosToVisit[i].y].OccupySpaces(b, startPos, _end);
381 dirToNextPlate = platePosToVisit[i + 1] - platePosToVisit[i];
382 }
383 else
384 {
385 allPlateArray[platePosToVisit[i].x, platePosToVisit[i].y].OccupySpaces(b, startPos, _end);
386 return platesToVisit;
387 }
388 if (dirToNextPlate.x == 0)
389 {//y bigger
390 if (dirToNextPlate.y > 0)
391 {
392 startPos.y = 0;
393 }
394 else
395 {
396 startPos.y = allPlateArray[platePosToVisit[i + 1].x, platePosToVisit[i + 1].y].Columns - 1;
397 }
398 }
399 else
400 {
401 if (dirToNextPlate.x > 0)
402 {
403 startPos.x = 0;
404 }
405 else
406 {
407 startPos.x = allPlateArray[platePosToVisit[i + 1].x, platePosToVisit[i + 1].y].Rows - 1;
408 }
409 }
410 }
411 }
412 else
413 {
414 allPlateArray[platePosToVisit.Last().x, platePosToVisit.Last().y].OccupySpaces(b, startPos, _end);
415 }
416 return platesToVisit;

418 void GenerateGoalNodes()
419 {
420 List<List<Bude>> groups = new List<List<Bude>>();
421 bool added = false;
422 int j = 0;
423 bool shouldBreak = false;
424 for (int i = 0; i < allBudenScripts!.Count; i++)
425 {
426 Bude goal = allBudenScripts[i];
427 while (j < groups.Count && !shouldBreak)
428 {
429 if (Vector3.Distance(groups[j][0].GetFarestPoint(), goal.GetFarestPoint()) <= maxGoalNodeDistance && WorldPositionToPlateArrayPosition(goal.GetFarestPoint()) == WorldPositionToPlateArrayPosition(groups[j][0].GetFarestPoint()))
430 {
431 Vector3 firstDirection = groups[j][0].GetFacingDirection();
432 Vector3 goalDirection = goal.GetFacingDirection();
433 float dot = Vector3.Dot(goalDirection.normalized, firstDirection.normalized);//dot product to get the direction the vectors are facing
434 //should divide by position
435 Vector3 displacement = groups[j][0].GetFarestPoint() - goal.GetFarestPoint();
436 float positionBudenToEachOther = Vector3.Dot(displacement.normalized, goalDirection.normalized);
437 if (positionBudenToEachOther > 0)
438 {
439 //only invalid position here are
440 //1: If it is not in front by enough
441 //2: They are pointing in the same direction and are directly infront of eathother(or to close)
442 if (positionBudenToEachOther > 0.5f) //everthing could be accetpable
443 {
444 if (positionBudenToEachOther > 0.8f)//only accetable if facing eachother
445 {
446 if (dot < -0.4f)
447 {
448 groups[j].Add(goal);
449 added = true;
450 shouldBreak = true;
451 }
452 }
453 else
454 {
455 groups[j].Add(goal);
456 added = true;
457 shouldBreak = true;
458 }
459 }
460 else // only accetpable if facing same direction
461 {
462 if (dot < 0.1f)
463 {
464 groups[j].Add(goal);
465 added = true;
466 shouldBreak = true;
467 }
468 }
469 }
470 else if (positionBudenToEachOther < 0)//the other but inverted
471 {
472 if (positionBudenToEachOther < -0.5f) //everthing could be accetpable
473 {
474 if (positionBudenToEachOther < -0.8f)//only accetable if facing eachother
475 {
476 if (dot < -0.4f)
477 {
478 groups[j].Add(goal);
479 added = true;
480 shouldBreak = true;
481 }
482 }
483 else
484 {
485 groups[j].Add(goal);
486 added = true;
487 shouldBreak = true;
488 }
489 }
490 else // only accetpable if facing same direction
491 {
492 if (dot > 0.9f)
493 {
494 groups[j].Add(goal);
495 added = true;
496 shouldBreak = true;
497 }
498 }
499 }
500 }
501 j++;
502 }
503 if (!added)
504 {
505 groups.Add(new List<Bude>());
506 groups[^1].Add(goal);
507 }
508 shouldBreak = false;
509 j = 0;
510 added = false;
511 }
512 foreach (List<Bude> goalGroup in groups)
513 {
514 GoalNode gn = new GoalNode(goalGroup, WorldPositionToPlate(goalGroup[0].GetFarestPoint()), this);
515 allGoalNodes.Add(gn);
516 gn.OnPlate.AddGoalNode(gn);
517 }

543 public Queue<Vector3> HandlePathRequest(Vector3 start, Vector3 goal)
544 {
545 if (allPositionsToGoals.ContainsKey((start, goal)))
546 {
547 if (PlatePathChanged(allPositionsToGoals[(start, goal)].Item1))
548 {
549 return new Queue<Vector3>(allPositionsToGoals[(start, goal)].Item2);
550 }
551 }
552 Vector2Int arrayStart = WorldPositionToPlateArrayPosition(start);
553 Vector2Int arrayGoal = WorldPositionToPlateArrayPosition(goal);
554 List<Vector2Int> platePosToVisit;
555 byte[,] flowField;
556 int[,] baseCostPlates;
557 List<Plate> platesToVisit;
558 if (goalPositionToFlowField.ContainsKey(goal))
559 {
560 baseCostPlates = goalPositionToFlowField[goal].Item1;
561 flowField = goalPositionToFlowField[goal].Item2;
562 }
563 else
564 {
565 baseCostPlates = GenerateMatrix.GenerateBaseCostMatrix(plateCountX, plateCountZ, (int row, int column) => !allPlateArray![row, column].HasOnlyObstacles, out bool onlyObstacles, out bool noObstacles);
566 (_, flowField) = GenerateMatrix.GenerateDistanceFieldAndFlowField(baseCostPlates, plateCountX, plateCountZ, new List<Vector2Int> { arrayGoal }, pathDiagonal);
567 }
568 platePosToVisit = GenerateMatrix.GetBestPathInFlowFieldFull(flowField, arrayStart);
569 if (platePosToVisit.Count == 0) { return new Queue<Vector3>(); }//there is no way to path to the goal from the given position
570 platesToVisit = new List<Plate>();
571 foreach (Vector2Int pos in platePosToVisit)
572 {
573 platesToVisit.Add(allPlateArray![pos.x, pos.y]);
574 }
575 (Queue<Vector3> wayPoints, Plate? lastVisitedPlate) = GenerateMatrix.GeneratePath(platesToVisit, start, goal, pathDiagonal);
576 if (lastVisitedPlate == null)
577 {
578 if (!goalPositionToFlowField.ContainsKey(goal))
579 {
580 goalPositionToFlowField.Add(goal, (baseCostPlates, flowField));
581 }
582 if (!allPositionsToGoals.ContainsKey((start, goal)))
583 {
584 allPositionsToGoals.Add((start, goal), (platesToVisit, new Queue<Vector3>(wayPoints)));
585 }
586 return wayPoints;
587 }
588 else
589 {
590 int tries = 0; // try to find a route 3 times else go to the exit
591 Plate goalPlate = allPlateArray![arrayGoal.x, arrayGoal.y];
592 do
593 {
594 Vector2Int platePos = WorldPositionToPlateArrayPosition(lastVisitedPlate!.Center);
595 baseCostPlates[platePos.x, platePos.y] = GenerateMatrix.MatrixObstacleValue;
596 (_, flowField) = GenerateMatrix.GenerateDistanceFieldAndFlowField(baseCostPlates, plateCountX, plateCountZ, new List<Vector2Int> { arrayGoal }, pathDiagonal);
597 platePosToVisit = GenerateMatrix.GetBestPathInFlowFieldFull(flowField, arrayStart);
598
599 if (platePosToVisit.Count == 0) { return new Queue<Vector3>(); }//there is no way to path to the goal from the given position
600
601 platesToVisit = new List<Plate>();
602 foreach (Vector2Int pos in platePosToVisit)
603 {
604 platesToVisit.Add(allPlateArray![pos.x, pos.y]);
605 }
606
607 (Queue<Vector3> newWayPoints, Plate? lastPlate) = GenerateMatrix.GeneratePath(platesToVisit, start, goal, pathDiagonal);
608 lastVisitedPlate = lastPlate;
609 if (lastPlate == null)
610 {
611 if (allPositionsToGoals.ContainsKey((start, goal)))
612 {
613 allPositionsToGoals.Remove((start, goal));
614 }
615 if (goalPositionToFlowField.ContainsKey(goal))
616 {
617 goalPositionToFlowField.Remove(goal);
618 }
619 allPositionsToGoals.Add((start, goal), (platesToVisit, new Queue<Vector3>(newWayPoints)));
620 goalPositionToFlowField.Add(goal, (baseCostPlates, flowField));
621 return newWayPoints;
622 }
623 tries++;
624 } while (tries < Mathf.Max(plateCountX, plateCountZ));
625 return new Queue<Vector3>();//no path could be found
626 }

646 public Plate WorldPositionToPlate(Vector3 pos)
647 {
648 int plateNumberX = Mathf.FloorToInt((pos.x - allFloorBounds[0].min.x) / normalPlateX);
649 int plateNumberZ = Mathf.FloorToInt((pos.z - allFloorBounds[0].min.z) / normalPlateZ);
650 plateNumberX = Math.Clamp(plateNumberX, 0, plateCountX - 1);
651 plateNumberZ = Math.Clamp(plateNumberZ, 0, plateCountZ - 1);
652 return allPlateArray![plateNumberX, plateNumberZ];
653 }
654 Vector2Int WorldPositionToPlateArrayPosition(Vector3 pos)
655 {
656 int plateNumberX = Mathf.FloorToInt((pos.x - allFloorBounds[0].min.x) / normalPlateX);
657 int plateNumberZ = Mathf.FloorToInt((pos.z - allFloorBounds[0].min.z) / normalPlateZ);
658 plateNumberX = Math.Clamp(plateNumberX, 0, plateCountX - 1);
659 plateNumberZ = Math.Clamp(plateNumberZ, 0, plateCountZ - 1);
660 return new Vector2Int(plateNumberX, plateNumberZ);
