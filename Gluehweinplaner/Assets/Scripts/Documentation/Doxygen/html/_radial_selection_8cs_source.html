150 public void GetSelectedRadialPart(float time)
151 {
152 int number = (isBude) ? numberOfHouseParts : ((left) ? numberOfLeftParts : numberOfRadialPart);
153 Vector3 centerToHand = ((left) ? handTransformLeft.position : handTransform.position) - radialPartCanvas.position;
154 Vector3 centerToHandProjected = Vector3.ProjectOnPlane(centerToHand, radialPartCanvas.forward);
155
156 float angle = Vector3.SignedAngle(radialPartCanvas.up, centerToHandProjected, -radialPartCanvas.forward);
157
158 if (angle < 0)
159 {
160 angle += 360;
161 }
162
163 //Debug.Log("ANGLE: " + angle);
164
165 currentSelectedRadialPart = (int)angle * number / 360;
166
167 // Play sound when selection changes
168 if (previousSelected != currentSelectedRadialPart)
169 {
170 SoundFXManager.instance.PlayRandomSoundFXClip(selectionChangeSoundClip, transform, 0.8f);
171 previousSelected = currentSelectedRadialPart;
172 }
173
174 if (currentSelectedRadialPart != waitingAt)
175 {
176 ResetWaitTimer(currentSelectedRadialPart);
177 }
178 timeWaited += time;
179
180 for (int i = 0; i < spawnedParts.Count; i++)
181 {
182 if (i == currentSelectedRadialPart)
183 {
184 spawnedParts[i].GetComponent<Image>().color = gradient.Evaluate(timeWaited / waitTimeUntilActivation);
185 spawnedParts[i].transform.localScale = 1.1f * UnityEngine.Vector3.one;
186 }
187 else
188 {
189 spawnedParts[i].GetComponent<Image>().color = Color.white;
190 spawnedParts[i].transform.localScale = 1 * UnityEngine.Vector3.one;
191 }
192 }
193 }

223 public void SpawnRadialPart()
224 {
225 Ray ray = new Ray(handTransform.position, handTransform.forward);
226 float maxDistance = 20f;
227 RaycastHit hit;
228 if (Physics.Raycast(ray, out hit, maxDistance))
229 {
230 if (hit.collider.gameObject.CompareTag("Bude"))
231 {
232 isBude = true;
233 selectedBude = hit.collider.transform.parent.gameObject.GetComponentInChildren<Bude>();
234 if (selectedBude)
235 {
236 Renderer r = hit.collider.GetComponent<Renderer>();
237 before = r.material;
238 r.material = highlightMaterial;
239 }
240 }
241 }
242 else
243 {
244 isBude = false;
245 selectedBude = null;
246
247 }
248
249 radialPartCanvas.gameObject.SetActive(true);
250
251 //play Placement Sound Effect
252 SoundFXManager.instance.PlaySoundFXClip(spawnRadialPartSoundClip, transform, 1f);
253
254 radialPartCanvas.position = (left) ? handTransformLeft.position : handTransform.position;
255 radialPartCanvas.position = (left) ? handTransformLeft.position : handTransform.position;
256 radialPartCanvas.rotation = (left) ? handTransformLeft.rotation : handTransform.rotation;
257
258 foreach (var item in spawnedParts)
259 {
260 Destroy(item);
261 }
262
263 spawnedParts.Clear();
264
265 int number = (isBude) ? numberOfHouseParts : numberOfRadialPart;
266 number = (left) ? numberOfLeftParts : number;
267
268 for (int i = 0; i < number; i++)
269 {
270
271 float angle = -i * 360 / number - angleBetweenPart / 2;
272 Vector3 radialPartEulerAngle = new Vector3(0, 0, angle);
273
274 GameObject spawnedRadialPart = Instantiate(radialPartPrefab, radialPartCanvas);
275 spawnedRadialPart.transform.position = radialPartCanvas.position;
276 spawnedRadialPart.transform.localEulerAngles = radialPartEulerAngle;
277
278 spawnedRadialPart.GetComponent<Image>().fillAmount = (1 / (float)number) - (angleBetweenPart / 360);
279 spawnedParts.Add(spawnedRadialPart);
280
281 TextMeshProUGUI buttonText = spawnedRadialPart.GetComponentInChildren<TextMeshProUGUI>();
282 int count = (isBude) ? partToFunctionHouse.Count : (left) ? partToFunctionLeft.Count : partToFunction.Count;
283 if (buttonText != null && i < count)
284 {
285 buttonText.text = (isBude) ? buttonLabelsHouse[i] : (left) ? buttonLabelsLeft[i] : buttonLabels[i];
286 }
287 }
